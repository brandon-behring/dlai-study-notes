    \node (input)  at (-2.6, 0) {$x$};
    \node (output) at (7.4, 0)  {$h = Wx + \dfrac{\alpha}{r}\,BAx$};

    \draw[arr] (input) -- (W);
    \draw[arr] (W) -- ++(1.6, 0);
    \draw[arr] (input) |- (A);
    \draw[arr] (A) -- (B);
    \draw[arr] (B) -| ++(1.6, 0.9);
