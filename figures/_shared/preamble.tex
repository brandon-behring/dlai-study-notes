% Shared TikZ preamble for DLAI Study Notes corpus.
%
% Every figure at figures/<book>/<name>.tex should \input this for
% consistent color palette, fonts, and arrow styles across all books.
%
% Compile mode:
%   \documentclass[tikz, border=8pt]{standalone}
%   \input{../_shared/preamble.tex}
%   \begin{document}
%   \begin{tikzpicture}
%     ...
%   \end{tikzpicture}
%   \end{document}
%
% Build pipeline (Phase 0.C.0 — to be added to book-scaffold v3.2):
%   pdflatex figures/<book>/<name>.tex  → figures/<book>/<name>.pdf
%   pdf2svg figures/<book>/<name>.pdf public/figures/<book>/<name>.svg
%
% Future-Brandon: this preamble starts minimal. Add corpus-consistent
% \tikzset definitions, custom node styles, and Warm Tol-5 hue palette
% references here once the first few figures define their needs.
%
% THEMING CONVENTION (so figures flip cleanly in dark mode):
%   - Labels/text/arrows: leave BLACK. build-figures remaps pure black →
%     var(--diagram-ink), so they flip to cream in dark and stay legible.
%   - Node identity: use COLORED BORDERS (draw=tolblue/tolgreen/...), NOT pale
%     tint fills. A light tint fill (e.g. tolblue!10) does NOT remap, so cream
%     text lands on a light box in dark mode → unreadable. Prefer transparent
%     fills + colored borders. (Saturated fills are fine; pale tints are the trap.)

\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}
\usetikzlibrary{arrows.meta, positioning, shapes, fit, calc, decorations.pathreplacing, backgrounds}

% Warm Tol 5-hue colorblind-safe palette (matches scaffold's Tol palette)
% reference: book-scaffold-astro/package/styles/tokens.css
\definecolor{tolblue}{HTML}{4477AA}
\definecolor{tolcyan}{HTML}{66CCEE}
\definecolor{tolgreen}{HTML}{228833}
\definecolor{tolyellow}{HTML}{CCBB44}
\definecolor{tolred}{HTML}{EE6677}
\definecolor{tolpurple}{HTML}{AA3377}
\definecolor{tolgrey}{HTML}{BBBBBB}

% Standard node + edge styles
\tikzset{
  >={Stealth[round]},
  every node/.style={font=\sffamily},
  node distance=1.2cm and 1.6cm,
  every edge/.style={draw, ->, semithick},
}
